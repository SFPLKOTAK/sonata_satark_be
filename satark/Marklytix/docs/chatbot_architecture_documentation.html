# Universal Entity Value Grounding & Typo-Resilient Matching in entity_grounding_engine.py
class MarklytixEntityGroundingEngine:
    def resolve_entities(self, query: str, candidate_tables: list = None, threshold: float = 0.78):
        # 1. Isolate contiguous content tokens (ignoring conversational stopwords)
        all_words = re.findall(r'\b\w+\b', query.lower())
        content_word_indices = [i for i, w in enumerate(all_words) if w not in generic_stopwords]
        
        # 2. Group contiguous words into candidate entity phrases (e.g. "ram devi")
        content_phrases = group_contiguous_tokens(all_words, content_word_indices)
        
        # 3. Non-Overlapping Longest-Match against Scoped Candidate Tables
        for group in content_phrases:
            phrase = " ".join([all_words[idx] for idx in group])
            match = self.fuzzy_match_against_registry(phrase, candidate_tables, threshold)
            if match:
                matched_results.append(match)
                break  # Consumes span, suppresses fragmented single-word noise
                
        return matched_results
# Graph Retrieval-Augmented Generation (GRAG) in graph_engine.py
class MarklytixRelationalGraph:
    def get_relational_join_blueprint(self, table_names: list):
        # 1. Discover multi-hop bridge tables via BFS shortest path
        connected_tables, bridges = self.discover_bridges_for_tables(table_names)
        
        # 2. Extract verified foreign-key & column-heuristic join relationships
        join_lines = []
        for t1, t2 in combinations(connected_tables, 2):
            joins = self.get_join_paths_between_tables(t1, t2)
            for j in joins:
                join_lines.append(f"-- dbo.[{t1}] JOIN dbo.[{t2}] ON dbo.[{t1}].[{j['key1']}] = dbo.[{t2}].[{j['key2']}]")
                
        return "\n".join(join_lines), bridges
# Negative Feedback Anti-Pattern Directives in consumers.py & views.py
def retrieve_dislike_feedback_warnings(self, query: str, k: int = 1):
    dislike_coll = client.get_collection(name="marklytix_dislike_feedback")
    results = dislike_coll.query(query_texts=[query], n_results=k)
    
    warnings = []
    for doc, meta, dist in zip(results["documents"][0], results["metadatas"][0], results["distances"][0]):
        if dist <= 1.05:  # Close semantic match to previously disliked query
            correction_note = meta.get("feedback_comments", "")
            warnings.append(f"-- ⚠️ CRITICAL WARNING (FROM USER DISLIKE FEEDBACK ON SIMILAR QUERY):")
            warnings.append(f"-- -> PREVENT THIS MISTAKE: {correction_note}")
            
    return "\n".join(warnings)
# Dynamic Column Pruning: Top 5% Join Frequency + LLM Metric Selector in consumers.py
def prune_table_schema(self, schema_text, user_query, table_name, max_columns=15):
    doc_meta = self.get_table_documentation_cache(self.engine).get(table_name.lower(), {})
    
    # 1. Mandatory Join Hubs: Primary Key + Top 5% Graph Frequency Join Columns
    mandatory_cols = set(doc_meta.get("heuristic_keys", []))
    mandatory_cols.add("id")
    
    # 2. LLM selects semantically relevant metric columns for the specific question
    prompt = f"""Given the user question: "{user_query}"
Table schema:
{schema_text}
Identify the top relevant columns needed to answer the question. Return a JSON array of column names."""
    
    selected_cols = self.query_model(prompt)
    final_cols = list(mandatory_cols.union(selected_cols))[:max_columns]
    return format_trimmed_schema(schema_text, final_cols)
# Dynamic Few-Shot SQL RAG Retrieval with Adaptive Distance Thresholding in consumers.py
def retrieve_top_k_sql_examples(self, query, subcategory=None, k=2):
    sql_coll = client.get_collection(name="marklytix_sql_examples")
    
    # 1. Subcategory search with automatic fallback to global semantic search
    res = sql_coll.query(query_texts=[query], n_results=k)
    
    valid_matches = []
    for doc, meta, dist in zip(res["documents"][0], res["metadatas"][0], res["distances"][0]):
        # Strict distance threshold (<= 1.05) eliminates unrelated domain examples
        if dist <= 1.05:
            valid_matches.append((doc, meta, dist))
            
    # Relative distance gap check (<= 0.25)
    if len(valid_matches) > 1:
        top_dist = valid_matches[0][2]
        valid_matches = [m for m in valid_matches if (m[2] - top_dist) <= 0.25]
        
    return format_few_shot_prompt(valid_matches)
# Level 4/5: Recursive SQL Validation & Self-Correction Loop in consumers.py
def execute_sql_with_self_correction(self, initial_query, user_question, max_retries=3):
    current_query = initial_query
    for attempt in range(max_retries):
        try:
            # Attempt read-only execution against SQL Server
            results_df = pd.read_sql_query(text(current_query), self.engine)
            return results_df, current_query, None
        except Exception as sql_err:
            print(f"⚠️ [Self-Correction] Attempt {attempt+1} failed: {sql_err}")
            # Feed error trace & original intent back to LLM for surgical correction
            correction_prompt = f"""The following T-SQL query failed on SQL Server:
```sql
{current_query}
```
Error Message: {str(sql_err)}
User Question: {user_question}
Fix the query. Return ONLY the valid bracketed T-SQL."""
            current_query = self.query_model(correction_prompt)
            
    raise Exception("Max self-correction retries exceeded.")
