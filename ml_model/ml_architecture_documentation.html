Centralizes feature column names, model save paths, and official grade cutoffs.

MODEL_FILE_PATH = ARTIFACTS_DIR / "audit_risk_model.joblib" CLASSIFIER_FILE_PATH = ARTIFACTS_DIR / "audit_grade_classifier.joblib" GRADE_THRESHOLDS = [ (70, 'A'), (60, 'B'), (50, 'C'), (0, 'D') ] FEATURE_COLS = [ 'Zone', 'Division', 'Region', 'PAR_1_30_Rate_Pct', 'Staff_Caseload_Ratio', 'Staff_Overstay_12M_Pct', 'Avg_Center_Size' ]
ml_model/feature_engineering.py Transformation

Converts raw counts into size-independent ratios and handles lag features.

def transform_features(df): # Size-independent ratios df['NPA_Rate_Pct'] = df['Cur_NPA_Amt'] * 100.0 / principal_os df['Staff_Caseload_Ratio'] = df['TotalLoans'] / df['ActiveStaffCount'] # Previous audit score lag df['Prev_Score'] = df.groupby('BRANCHID')['Score'].shift(1) # Categorical hierarchy for cat in ['Zone', 'Division', 'Region']: df[cat] = df[cat].astype('category') return df
ml_model/ensemble.py Stacking Engine

Trains Stacking Ensemble (XGBoost + CatBoost + LightGBM + HistGB) with Ridge Meta-Learner.

def train_stacking_ensemble(): stacking_reg = StackingRegressor( estimators=[('xgb', XGBRegressor()), ('cat', CatBoostRegressor()), ('lgb', LGBMRegressor())], final_estimator=Ridge(alpha=1.0) ) stacking_reg.fit(X_train, y_score_train)
ml_model/predict.py Inference API

Instant prediction engine with risk warning triggers for Django views.

def predict_branch_audit_score(branch_data): regressor, classifier, label_encoder = get_models() X = transform_features(branch_data)[FEATURE_COLS] score_pred = regressor.predict(X)[0] grade_pred = label_encoder.inverse_transform(classifier.predict(X))[0] return { 'predicted_score': score_pred, 'predicted_grade': grade_pred, 'risk_level': 'HIGH' if grade_pred in ['C', 'D'] else 'NORMAL' }
